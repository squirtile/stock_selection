# ── 股票选股 ──

0 13 * * 1-5 /usr/bin/flock -n /tmp/stock_intraday_scanner.lock /usr/bin/timeout 7200 /bin/bash -lc 'cd /home/ubuntu/code/stock/stock_selection && PYTHONUNBUFFERED=1 /usr/bin/python3 intraday_scanner.py --loop --interval 60 >> output/intraday_cron.log 2>&1'
# 收盘后只用当天免费网页快照评价“个股+板块”过滤效果，不请求Tushare/BaoStock。
10 15 * * 1-5 cd /home/ubuntu/code/stock/stock_selection && /usr/bin/python3 backtest/intraday_sector_review.py >> output/intraday_sector_review.log 2>&1

30 20 * * 1-5 cd /home/ubuntu/code/stock/stock_selection && /usr/bin/python3 backtest/intraday_sector_historical_research.py >> output/intraday_sector_historical.log 2>&1

# 每个交易日 18:30 执行每日选股报告。分钟线只更新前 200 只，避免全市场分钟缓存
# 阻塞整批日线选股结果；日线仍全量强制更新。flock 防重入，timeout 防止异常任务长期占用。
30 18 * * 1-5 /usr/bin/flock -n /tmp/stock_daily_report.lock /usr/bin/timeout 7200 /bin/bash -lc 'cd /home/ubuntu/code/stock/stock_selection && PYTHONUNBUFFERED=1 /usr/bin/python3 daily_report.py --force-update --minute-max-stocks 200 --no-email >> output/daily_report_cron.log 2>&1'
